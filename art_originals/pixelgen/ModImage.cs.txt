// The mod image (SpireBrothers/mod_image.png, 420 x 420): the four brothers in a 2 x 2 grid, no names.
// Each tile uses its brother's select-button colors (sampled from art_originals/select_icons_old) with rings behind the face.
// Use it like SelectButtons.cs.txt: swap it in as Program.cs, run, then copy out/mod_image.png into SpireBrothers/.
var P = "D:/projects/csharp/SpireBrothers/art_originals/select_icons_old/";
AnimatedBrother[] brothers = [new DanielArt(), new DavidArt(), new JoshuaArt(), new TimArt()];
const int Size = 420, Tile = 210, Gap = 4;
var img = new Canvas(Size, Size);
for (int y = 0; y < Size; y++) for (int x = 0; x < Size; x++) img.Set(x, y, Col.Hex("16141c"));

for (int i = 0; i < brothers.Length; i++)
{
    var b = brothers[i];
    b.Portrait = true;
    var frame = b.Render(b.Anims()["idle"].poses[0]);
    var orig = PngRead.Load(P + $"{b.Name}/select.png");
    Col top = orig.Get(0, 0)!.Value, low = orig.Get(0, 150)!.Value;
    int ox = (i % 2) * Tile, oy = (i / 2) * Tile;
    int x0 = ox + (i % 2 == 0 ? 0 : Gap / 2), x1 = ox + Tile - (i % 2 == 0 ? Gap / 2 : 0);
    int y0 = oy + (i / 2 == 0 ? 0 : Gap / 2), y1 = oy + Tile - (i / 2 == 0 ? Gap / 2 : 0);
    float cx = ox + Tile / 2f, cy = oy + Tile * 0.42f;

    // gradient plus soft rings around the face, like the select buttons
    for (int y = y0; y < y1; y++) for (int x = x0; x < x1; x++)
    {
        var g = top.Lerp(low, (y - y0) / (float)(y1 - y0));
        float r = MathF.Sqrt((x + 0.5f - cx) * (x + 0.5f - cx) + (y + 0.5f - cy) * (y + 0.5f - cy));
        float m = r % 26f;
        float ring = Math.Clamp(Math.Min(m - 20f, 26f - m + 0.5f), 0, 1);
        img.Set(x, y, g.Lerp(new Col(255, 255, 255), ring * 0.12f));
    }

    // portrait: head and shoulders (frame x 8..51, y 2..51) at 4x, centered, with a drop shadow, cut off at the tile bottom
    const int S = 4, X0 = 8, Y0 = 2, CW = 44, CH = 50;
    int px = (int)(cx - CW * S / 2f), py = y1 - CH * S + 6;
    Col? Port(int x, int y) => frame.Get(X0 + x / S, Y0 + y / S);
    for (int pass = 0; pass < 2; pass++)
        for (int y = 0; y < CH * S; y++) for (int x = 0; x < CW * S; x++)
        {
            if (Port(x, y) is not { } c) continue;
            int tx = px + x + (pass == 0 ? 5 : 0), ty = py + y + (pass == 0 ? 5 : 0);
            if (tx < x0 || tx >= x1 || ty < y0 || ty >= y1) continue;
            img.Set(tx, ty, pass == 0 ? img.Get(tx, ty)!.Value.Lerp(new Col(0, 0, 0), 0.35f) : c);
        }
}
img.SavePng("out/mod_image.png");
img.SavePng("out/mod_image_preview.png", 2);
Console.WriteLine("out/mod_image.png");
